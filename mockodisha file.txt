        }
      }
      renderQuestion();
      renderPalette();
    }

    function navigateQuestion(dir) {
      const newIdx = currentQuestion + dir;
      if (newIdx >= 0 && newIdx < questionsData.length) {
        currentQuestion = newIdx;
        if (questionStates[currentQuestion] === 'not-visited') {
          questionStates[currentQuestion] = 'not-answered';
        }
        renderQuestion();
        renderPalette();
      }
    }

    function jumpToQuestion(idx) {
      currentQuestion = idx;
      if (questionStates[currentQuestion] === 'not-visited') {
        questionStates[currentQuestion] = 'not-answered';
      }
      renderQuestion();
      renderPalette();
    }

    function renderPalette() {
      const grid = document.getElementById('palette-grid');
      grid.innerHTML = '';

      questionsData.forEach((q, idx) => {
        const btn = document.createElement('button');
        const st = questionStates[idx];
        btn.className = `palette-btn status-${st} ${idx === currentQuestion ? 'active' : ''}`;
        btn.innerText = idx + 1;
        btn.onclick = () => jumpToQuestion(idx);
        grid.appendChild(btn);
      });

      updateCounts();
    }

    function updateCounts() {
      let visited = 0, notAnswered = 0, answered = 0, marked = 0, ansMarked = 0;
      questionStates.forEach(st => {
        if (st === 'not-visited') visited++;
        else if (st === 'not-answered') notAnswered++;
        else if (st === 'answered') answered++;
        else if (st === 'marked') marked++;
        else if (st === 'ans-marked') ansMarked++;
      });

      document.getElementById('cnt-visited').innerText = visited;
      document.getElementById('cnt-not-answered').innerText = notAnswered;
      document.getElementById('cnt-answered').innerText = answered;
      document.getElementById('cnt-marked').innerText = marked;
      document.getElementById('cnt-ans-marked').innerText = ansMarked;
    }

    function startTimer() {
      timerInterval = setInterval(() => {
        if (timeLeft <= 0) {
          clearInterval(timerInterval);
          submitExam();
          return;
        }
        timeLeft--;
        const mins = Math.floor(timeLeft / 60);
        const secs = timeLeft % 60;
        document.getElementById('timer-display').innerText = 
          `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
      }, 1000);
    }

    function toggleMobileSidebar() {
      const sidebar = document.getElementById('sidebar');
      const backdrop = document.getElementById('sidebar-backdrop');
      sidebar.classList.toggle('open');
      backdrop.classList.toggle('show');
    }

    function toggleFullScreen() {
      if (!document.fullscreenElement) {
        document.documentElement.requestFullscreen().catch(() => {});
      } else {
        if (document.exitFullscreen) document.exitFullscreen();
      }
    }

    function submitExam() {
      if (confirm("Are you sure you want to submit the exam?")) {
        isExamActive = false;
        clearInterval(timerInterval);
        let correct = 0;
        let wrong = 0;
        let detailsHTML = '';

        questionsData.forEach((q, idx) => {
          const userAns = userAnswers[idx];
          let statusClass = 'unanswered';

          if (userAns !== null) {
            if (userAns === q.correctAnswer) {
              correct++;
              statusClass = 'correct';
            } else {
              wrong++;
              statusClass = 'incorrect';
            }
          }

          detailsHTML += `
            <div class="result-item ${statusClass}">
              <div class="result-q-title">Q${q.id}. ${q.question}</div>
              <div style="font-size:0.8rem; color:#475569;">
                Your Answer: <strong>${userAns !== null ? q.options[userAns] : 'Not Answered'}</strong> | 
                Correct Answer: <strong style="color:#10b981;">${q.options[q.correctAnswer]}</strong>
              </div>
            </div>
          `;
        });

        const score = (correct * 1.0) - (wrong * 0.25);
        document.getElementById('res-score').innerText = `${score.toFixed(2)} / 100`;
        document.getElementById('res-correct').innerText = correct;
        document.getElementById('res-wrong').innerText = wrong;
        document.getElementById('results-detail-list').innerHTML = detailsHTML;

        document.getElementById('result-modal').classList.add('show');
      }
    }
